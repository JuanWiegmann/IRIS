    \node[font=\normalsize\itshape\color{gray}, above=0.5cm of userllm, text width=10cm, align=center]

    % Anleitung (instructions the LLM follows)
    \node[anleitung, left=3cm of userllm] (anleitung)
        {Anleitung\\{\small\color{gray} Protocol the LLM follows}\\{\small\color{gray} (tool descriptions + rules)}};

    \draw[arrowgreen] (anleitung.east) -- node[lbl, above] {Guides behavior} (userllm.west);

    % ════════════════════════════════════════════════════════════
    %  MCP TOOLS (KIM's exposed interface)
    % ════════════════════════════════════════════════════════════

    % Tool row
    \node[mcptool, below=3cm of userllm, xshift=-7cm] (t_context)
        {get\_context(query)\\{\small\color{gray} profile + relevant examples}};
    \node[mcptool, right=1cm of t_context] (t_check)
        {check\_draft(draft)\\{\small\color{gray} validate against profile}};
    \node[mcptool, right=1cm of t_check] (t_onboard)
        {onboard(answer)\\{\small\color{gray} GATE preference capture}};
    \node[mcptool, right=1cm of t_onboard] (t_log)
        {log\_output(text)\\{\small\color{gray} store for future retrieval}};

    % LLM → Tools (numbered steps)
    \node[step, below=1.2cm of userllm, xshift=-3cm] (s1) {1};
    \node[step, below=1.2cm of userllm, xshift=0cm] (s2) {3};
    \node[step, below=1.2cm of userllm, xshift=3cm] (s5) {5};

    \draw[arrowmcp] (userllm.south) -- ++(0, -0.5) -| (t_check.north);
    \draw[arrowmcp] (userllm.south) -- ++(0, -0.5) -| (t_onboard.north);
    \draw[arrowmcp] (userllm.south) -- ++(0, -0.5) -| (t_log.north);

    %  THE FLOW (numbered, right side)
    % ════════════════════════════════════════════════════════════

    \node[anchor=north west, font=\normalsize, text width=7.5cm, align=left]
        at (9, 0) (flow) {
        \textbf{\large Interaction Flow}\\[8pt]
        \textbf{\color{blue}1.} User writes message to their LLM\\[4pt]
        \textbf{\color{accent}2.} LLM calls \texttt{get\_context(query)}\\
        \quad\quad KIM returns profile + relevant examples\\[4pt]
        \textbf{\color{blue}3.} LLM generates draft response\\
        \quad\quad (using KIM's context)\\[4pt]
        \textbf{\color{accent}4.} LLM calls \texttt{check\_draft(draft)}\\
        \quad\quad KIM validates against profile\\
        \quad\quad \textit{\color{gray}(LLM doesn't know it's checking itself)}\\[4pt]
        \textbf{\color{blue}5.} LLM revises if needed\\[4pt]
        \textbf{\color{accent}6.} LLM calls \texttt{log\_output(final)}\\
        \quad\quad KIM stores for future retrieval\\[12pt]
        \textbf{\color{green}Key insight:}\\
        The LLM thinks \texttt{check\_draft} is\\
        an independent external service.\\
        It gets unbiased feedback because\\
        KIM validates with fresh eyes\\
        (profile rules only, no generation context).
    };

    % ════════════════════════════════════════════════════════════
    %  LAYER 2: SAMPLING (advanced, if supported)
    % ════════════════════════════════════════════════════════════

        at (9, -8.5) (layer2) {
        \textbf{\large Layer 2: MCP Sampling}\\
        {\small\color{gray}(Claude and advanced clients only)}\\[8pt]
        If the LLM supports MCP sampling,\\
        KIM can orchestrate multi-step flows:\\[4pt]
        \textbullet\ Fresh context per ``agent'' role\\
        \textbullet\ KIM controls the loop\\
        \textbullet\ Better isolation between steps\\
        \textbullet\ Same user LLM does the compute\\[8pt]
        Falls back to Layer 1 (Anleitung)\\
        for simpler MCP clients.
    };
